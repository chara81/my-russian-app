    // 日付から固定の問題5個を取得する簡易ハッシュ関数

    function getDailyData(dateStr) {

      let seed = 0;

      for (let i = 0; i < dateStr.length; i++) {

        seed += dateStr.charCodeAt(i);

      }

     

      const sentences = [];

      const words = [];

      for (let i = 0; i < 5; i++) {

        sentences.push(SENTENCES_POOL[(seed + i * 3) % SENTENCES_POOL.length]);

        words.push(WORDS_POOL[(seed + i * 2) % WORDS_POOL.length]);

      }

      return { sentences, words };

    }
